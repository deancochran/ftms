cmake_minimum_required(VERSION 3.16)
project(ftms_conan_consumer LANGUAGES C CXX)
find_package(ftms CONFIG REQUIRED)
add_executable(test_package test_package.c)
target_link_libraries(test_package PRIVATE ftms::ftms)
add_executable(test_package_cpp test_package.cpp)
target_link_libraries(test_package_cpp PRIVATE ftms::ftms)
target_compile_features(test_package_cpp PRIVATE cxx_std_11)
enable_testing()
add_test(NAME c_consumer COMMAND test_package)
add_test(NAME cpp_consumer COMMAND test_package_cpp)
